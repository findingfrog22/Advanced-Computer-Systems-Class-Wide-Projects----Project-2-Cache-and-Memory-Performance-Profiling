root@timothy-inspiron73062n1:~# sudo bash -c '
# Get current user ID and shell cgroup path
UID_VAL=$(id -u)
SHELL_CG=$(cut -d: -f3 /proc/$$/cgroup)

echo "--- Clearing cgroup cpuset limits ---"

# Step 1: Force reset user.slice and user@UID.service via systemd
systemctl set-property user-${UID_VAL}.slice AllowedCPUs= 2>/dev/null || true
systemctl set-property user@${UID_VAL}.service AllowedCPUs= 2>/dev/null || true

# Step 2: Iterate through every nested directory in /sys/fs/cgroup and write max/0-7
find /sys/fs/cgroup/user.slice -name "cpuset.cpus" 2>/dev/null | while read -r line; do
    echo "0-7" > "$line" 2>/dev/null || echo "max" > "$line" 2>/dev/null || true 
done

echo "Done."
'
--- Clearing cgroup cpuset limits ---
Done.
